#version 460 core

in vec3 FragPos;
in vec3 Normal;
in vec2 TexCoord;

out vec4 FragColor;

uniform sampler2D texture_main;
uniform int draw_texture;
uniform float alpha;

void main()
{
    vec3 lightDir = normalize(gl_LightSource[0].position.xyz - FragPos);   
    vec3 viewDir = normalize(-FragPos); 
    vec3 reflectDir = reflect(-lightDir, Normal);  

    vec4 ambient = gl_LightModel.ambient;
    vec4 diffuse = gl_LightSource[0].diffuse * max(dot(Normal, lightDir), 0.0);
    vec4 specular = gl_LightSource[0].specular * pow(max(dot(reflectDir, viewDir), 0.0), gl_FrontMaterial.shininess);

    vec4 lighting = ambient + diffuse + specular;

    if (draw_texture == 1)
    {
        vec4 texColor = texture(texture_main, TexCoord);
        FragColor = vec4(texColor.xyz, alpha);
    }
    else
    {
        FragColor = lighting;
    }
}
